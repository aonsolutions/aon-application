    it('Telefonica', done => {

			fs.createReadStream('test/pdfs/telefonica.zip')
			.pipe(unzip.Extract({ path: 'test/pdfs' }))
			.on('close', () =>
				fs.readdir('./test/pdfs/telefonica', function (err, files) {
				    //handling error
				    if (err) {
				        return done('Unable to scan directory: ' + err);
				    }
				    //listing all files using forEach
				    Promise.all(
							files
							.map( file => `./test/pdfs/telefonica/${file}`)
							.filter( file => "application/pdf" == fileType(readChunk.sync(file, 0, fileType.minimumBytes)).mime )
							.map( file =>  {
				        // Do whatever you want to do with the file
								return new Promise((resolve, reject) =>
									pdfParser.extractFromFile(
										file
										, (err, invoice) => {
											if ( err ) {
												// resolve(invoice);
												reject(`${file}: ${err}`);
											}
											else {
												// console.log(file);
												resolve(invoice);
											}
										}
									)
								);
				    }))
						.then(invoices => done() )
						.catch( err => done(err) )
						;
				})
			)
    });
